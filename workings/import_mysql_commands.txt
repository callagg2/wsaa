import mysql.connector

db = mysql.connector.connect(
host = "localhost",

password = "???"
database = "???"
)

cursor = db.cursor() # this is where the data is stored when read from the database

values = 1315

cursor.execute(sql, values) # you are executing sql on the data that has been read from the database
